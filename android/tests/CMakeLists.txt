cmake_minimum_required(VERSION 3.16)
project(ProKritaGeometryTests LANGUAGES CXX)
enable_testing()
add_executable(prokrita_geometry_test geometry_test.cpp)
target_include_directories(prokrita_geometry_test PRIVATE ../native)
target_compile_features(prokrita_geometry_test PRIVATE cxx_std_17)
add_test(NAME adaptive_geometry COMMAND prokrita_geometry_test)
add_executable(prokrita_gap_fill_test gap_fill_test.cpp)
target_include_directories(prokrita_gap_fill_test PRIVATE ../native)
target_compile_features(prokrita_gap_fill_test PRIVATE cxx_std_17)
add_test(NAME bounded_gap_fill COMMAND prokrita_gap_fill_test)
option(PROKRITA_WIDGET_TESTS "Build offscreen Qt canvas widget checks" OFF)
if(PROKRITA_WIDGET_TESTS)
    find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
    add_executable(prokrita_canvas_widgets_test canvas_widgets_test.cpp)
    target_include_directories(prokrita_canvas_widgets_test PRIVATE ../native)
    target_compile_features(prokrita_canvas_widgets_test PRIVATE cxx_std_17)
    target_link_libraries(prokrita_canvas_widgets_test PRIVATE Qt5::Widgets Qt5::Test)
    add_test(NAME canvas_widget_interactions COMMAND prokrita_canvas_widgets_test)
    set_tests_properties(canvas_widget_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
